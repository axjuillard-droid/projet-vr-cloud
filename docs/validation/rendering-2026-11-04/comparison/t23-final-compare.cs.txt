if(UnityEditor.EditorApplication.isPlaying||UnityEditor.EditorApplication.isCompiling)return "Refus : éditeur occupé";
for(int i=0;i<UnityEngine.SceneManagement.SceneManager.sceneCount;i++)if(UnityEngine.SceneManagement.SceneManager.GetSceneAt(i).isDirty)return "Refus : scène modifiée";
var previous=UnityEngine.SceneManagement.SceneManager.GetActiveScene().path;
var scene=UnityEditor.SceneManagement.EditorSceneManager.OpenScene("Assets/Scenes/DataCenter.unity");
var legacyLights=new System.Collections.Generic.List<Light>();
var oldPipeline=QualitySettings.renderPipeline;var temporaryPipeline=UnityEngine.Object.Instantiate(oldPipeline);var msaa=temporaryPipeline.GetType().GetProperty("msaaSampleCount");
var folder=System.IO.Path.GetFullPath("../tmp/unity/t23-comparison");System.IO.Directory.CreateDirectory(folder);
var snapshots=new System.Collections.Generic.Dictionary<Renderer,Material[]>();
var clones=new System.Collections.Generic.List<Material>();
var lightTypes=new System.Collections.Generic.Dictionary<Light,LightType>();
var rotations=new System.Collections.Generic.Dictionary<Light,Quaternion>();
var angles=new System.Collections.Generic.Dictionary<Light,Vector2>();
var cameraGo=new GameObject("T23_Camera_TEMP");var camera=cameraGo.AddComponent<Camera>();camera.enabled=false;camera.nearClipPlane=.05f;camera.farClipPlane=1000;camera.fieldOfView=65;
var target=new RenderTexture(1280,720,24);var pixels=new Texture2D(1280,720,TextureFormat.RGB24,false);camera.targetTexture=target;var active=RenderTexture.active;
var names=new[]{"ceiling","floor","network","distance"};
var positions=new[]{new Vector3(0,1.68f,8),new Vector3(0,1.68f,8),new Vector3(4,1.68f,6),new Vector3(0,1.68f,8)};
var views=new[]{Quaternion.Euler(-28,0,0),Quaternion.Euler(34,0,0),Quaternion.Euler(0,45,0),Quaternion.identity};
System.Action<string> capture=delegate(string variant){for(int i=0;i<names.Length;i++){camera.transform.SetPositionAndRotation(positions[i],views[i]);camera.Render();RenderTexture.active=target;pixels.ReadPixels(new Rect(0,0,1280,720),0,0);pixels.Apply();System.IO.File.WriteAllBytes(System.IO.Path.Combine(folder,variant+"-"+names[i]+".png"),pixels.EncodeToPNG());}};
int count=0;var materials=new System.Collections.Generic.List<object>();
try{
    // Base et variante matériaux déjà conservées.
    foreach(var renderer in UnityEngine.Object.FindObjectsByType<Renderer>(FindObjectsSortMode.None)){
        var original=renderer.sharedMaterials;var changed=(Material[])original.Clone();bool replace=false;
        for(int i=0;i<original.Length;i++){
            var mat=original[i];if(mat==null||(mat.name!="WS_FloorTile"&&mat.name!="WS_FloorPerf"&&mat.name!="WS_Ceiling"))continue;
            materials.Add(new {name=mat.name,metallic=mat.GetFloat("_Metallic"),smoothness=mat.GetFloat("_Smoothness"),shader=mat.shader.name});
            var clone=new Material(mat);clones.Add(clone);clone.SetFloat("_Metallic",0);clone.SetFloat("_Smoothness",mat.name=="WS_Ceiling"?.15f:.3f);changed[i]=clone;replace=true;
        }
        if(replace){snapshots.Add(renderer,original);renderer.sharedMaterials=changed;}
    }
    // Garder seulement les nouvelles variantes.
    foreach(var light in UnityEngine.Object.FindObjectsByType<Light>(FindObjectsSortMode.None))if(light.name=="Eclairage_WhiteSpace"){
        lightTypes.Add(light,light.type);rotations.Add(light,light.transform.rotation);angles.Add(light,new Vector2(light.spotAngle,light.innerSpotAngle));
        light.type=LightType.Spot;light.spotAngle=110;light.innerSpotAngle=80;light.transform.rotation=Quaternion.Euler(90,0,0);count++;
    }
    foreach(var l in UnityEngine.Object.FindObjectsByType<Light>(FindObjectsSortMode.None))if(l.name=="Eclairage_Allee"||l.name=="Eclairage_Lateral"){legacyLights.Add(l);l.enabled=false;}
    capture("final1x");
    msaa.SetValue(temporaryPipeline,4,null);QualitySettings.renderPipeline=temporaryPipeline;camera.targetTexture=null;target.Release();target.antiAliasing=4;target.Create();camera.targetTexture=target;
    capture("final4x");
    return new {passed=true,views=8,width=1280,height=720,lights=count,materials=materials.ToArray(),quality=QualitySettings.names[QualitySettings.GetQualityLevel()],graphics=SystemInfo.graphicsDeviceType.ToString(),source="Camera.Render outside Play, no Overlay or performance measurement"};
}finally{
    QualitySettings.renderPipeline=oldPipeline;UnityEngine.Object.DestroyImmediate(temporaryPipeline);foreach(var l in legacyLights)l.enabled=true;
    foreach(var entry in snapshots)entry.Key.sharedMaterials=entry.Value;
    foreach(var entry in lightTypes){entry.Key.type=entry.Value;entry.Key.transform.rotation=rotations[entry.Key];entry.Key.spotAngle=angles[entry.Key].x;entry.Key.innerSpotAngle=angles[entry.Key].y;}
    foreach(var clone in clones)UnityEngine.Object.DestroyImmediate(clone);
    RenderTexture.active=active;camera.targetTexture=null;UnityEngine.Object.DestroyImmediate(cameraGo);UnityEngine.Object.DestroyImmediate(pixels);target.Release();UnityEngine.Object.DestroyImmediate(target);
    UnityEditor.SceneManagement.EditorSceneManager.OpenScene(previous);
}
